DELIMITER $$

CREATE PROCEDURE IssueBook(
    IN p_BookID INT,
    IN p_BorrowerName VARCHAR(100)
)
BEGIN
    DECLARE book_status VARCHAR(20);

    -- Check book status
    SELECT Status INTO book_status
    FROM Books
    WHERE BookID = p_BookID;

    IF book_status = 'Available' THEN
        
        -- Insert transaction
        INSERT INTO Transactions(BookID, BorrowerName, IssueDate)
        VALUES(p_BookID, p_BorrowerName, CURDATE());

        -- Update book status
        UPDATE Books
        SET Status = 'Issued'
        WHERE BookID = p_BookID;

        SELECT 'Book Issued Successfully' AS Message;
    
    ELSE
        SELECT 'Book is Not Available' AS Message;
    END IF;

END $$

DELIMITER ;


CALL IssueBook(102, 'Riya Sharma');

DELIMITER $$

CREATE PROCEDURE ReturnBook(
    IN p_BookID INT
)
BEGIN
    DECLARE book_status VARCHAR(20);
    DECLARE latest_transaction INT;

    -- Check status
    SELECT Status INTO book_status
    FROM Books
    WHERE BookID = p_BookID;

    IF book_status = 'Issued' THEN
        
        -- Get latest transaction
        SELECT TransactionID INTO latest_transaction
        FROM Transactions
        WHERE BookID = p_BookID
        ORDER BY IssueDate DESC
        LIMIT 1;

        -- Delete latest transaction
        DELETE FROM Transactions
        WHERE TransactionID = latest_transaction;

        -- Update book status
        UPDATE Books
        SET Status = 'Available'
        WHERE BookID = p_BookID;

        SELECT 'Book Returned Successfully' AS Message;

    ELSE
        SELECT 'Book is Already Available' AS Message;
    END IF;

END $$

DELIMITER ;

CALL ReturnBook(101);

DELIMITER $$

CREATE PROCEDURE CountIssuedBooks(
    OUT totalIssued INT
)
BEGIN
    SELECT COUNT(*)
    INTO totalIssued
    FROM Books
    WHERE Status = 'Issued';
END $$

DELIMITER ;

CALL CountIssuedBooks(@count);
SELECT @count AS Total_Issued_Books;

CREATE PROCEDURE GetBorrowedBooks(
    IN p_BorrowerName VARCHAR(100)
)
BEGIN
    SELECT 
        B.Title,
        B.Author,
        T.IssueDate
    FROM Books B
    JOIN Transactions T
        ON B.BookID = T.BookID
    WHERE T.BorrowerName = p_BorrowerName;
END $$

DELIMITER ;

CALL GetBorrowedBooks('Arjun Varma');


(e).
DELIMITER $$

CREATE PROCEDURE CalculateFine(
    IN p_BookID INT,
    INOUT FineAmount INT
)
BEGIN
    DECLARE borrowed_days INT;
    DECLARE issue_date DATE;

    -- Get latest issue date
    SELECT IssueDate INTO issue_date
    FROM Transactions
    WHERE BookID = p_BookID
    ORDER BY IssueDate DESC
    LIMIT 1;

    -- Calculate borrowed days
    SET borrowed_days = DATEDIFF(CURDATE(), issue_date);

    IF borrowed_days > 7 THEN
        SET FineAmount = (borrowed_days - 7) * 5;
    ELSE
        SET FineAmount = 0;
    END IF;

    SELECT 
        p_BookID AS BookID,
        borrowed_days AS Borrowed_Days,
        FineAmount AS Fine_Amount;
END $$

DELIMITER ;


SET @fine = 0;
CALL CalculateFine(101, @fine);
SELECT @fine AS Total_Fine;
