CREATE TABLE Employee_Audit (
    ssn CHAR(9),
    action VARCHAR(20),
    changed_at TIMESTAMP
);

DELIMITER $$

CREATE TRIGGER after_employee_insert
AFTER INSERT ON EMPLOYEE
FOR EACH ROW
BEGIN
    INSERT INTO Employee_Audit(ssn, action, changed_at)
    VALUES (NEW.Ssn, 'INSERT', NOW());
END$$

DELIMITER ;

DELIMITER $$

CREATE TRIGGER prevent_salary_decrease
BEFORE UPDATE ON EMPLOYEE
FOR EACH ROW
BEGIN
    IF NEW.Salary < OLD.Salary THEN
        SET NEW.Salary = OLD.Salary;
    END IF;
END$$

DELIMITER ;

DELIMITER $$

CREATE TRIGGER check_salary_limit
BEFORE INSERT ON EMPLOYEE
FOR EACH ROW
BEGIN
    IF NEW.Salary > 60000 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Error: Salary exceeds maximum department limit.';
    END IF;
END$$

CREATE TABLE Department_Stats (
    Dnumber INT PRIMARY KEY,
    Total_Salary_Expenditure DECIMAL(10,2)
);

DELIMITER $$

CREATE TRIGGER update_salary_after_delete
AFTER DELETE ON EMPLOYEE
FOR EACH ROW
BEGIN
    UPDATE Department_Stats
    SET Total_Salary_Expenditure =
        Total_Salary_Expenditure - OLD.Salary
    WHERE Dnumber = OLD.Dno;
END$$

DELIMITER ;
