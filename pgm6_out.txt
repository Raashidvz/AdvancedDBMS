DELIMITER //

CREATE FUNCTION CalculateBonus (salary DECIMAL(10,2))
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    DECLARE bonus DECIMAL(10,2);
    SET bonus = salary * 0.10;
    RETURN bonus;
END //

DELIMITER ;


SELECT CalculateBonus(30000);

select Fname, Lname, Salary, CalculateBonus(salary) as BONUS FROM EMPLOYEE;

DELIMITER //

CREATE FUNCTION CalculateGrade (mark INT)
RETURNS VARCHAR(2)
DETERMINISTIC
BEGIN
    DECLARE grade VARCHAR(2);

    IF mark >= 90 THEN
        SET grade = 'S';
    ELSEIF mark >= 85 THEN
        SET grade = 'A+';
    ELSEIF mark >= 80 THEN
        SET grade = 'A';
    ELSEIF mark >= 75 THEN
        SET grade = 'B+';
    ELSEIF mark >= 70 THEN
        SET grade = 'B';
    ELSEIF mark >= 65 THEN
        SET grade = 'C+';
    ELSEIF mark >= 60 THEN
        SET grade = 'C';
    ELSEIF mark >= 55 THEN
        SET grade = 'D';
    ELSEIF mark >= 50 THEN
        SET grade = 'P';
    ELSE
        SET grade = 'F';
    END IF;

    RETURN grade;
END //

DELIMITER ;


SELECT CalculateGrade(87);

DELIMITER //

CREATE FUNCTION TaxCalculator (emp_ssn CHAR(9))
RETURNS DECIMAL(12,2)
DETERMINISTIC
BEGIN
    DECLARE monthly_salary DECIMAL(10,2);
    DECLARE annual_salary DECIMAL(12,2);
    DECLARE tax DECIMAL(12,2);

    -- Get monthly salary
    SELECT Salary INTO monthly_salary
    FROM EMPLOYEE
    WHERE Ssn = emp_ssn;

    -- Calculate annual salary
    SET annual_salary = monthly_salary * 12;

    -- Calculate tax
    IF annual_salary > 500000 THEN
        SET tax = annual_salary * 0.20;
    ELSEIF annual_salary >= 300000 THEN
        SET tax = annual_salary * 0.10;
    ELSE
        SET tax = 0;
    END IF;

    RETURN tax;
END //

DELIMITER ;


SELECT Fname, Lname, Salary as Salary_per_month, (Salary*12) as Annual_salary, TaxCalculator(Ssn) as Tax FROM EMPLOYEE;
